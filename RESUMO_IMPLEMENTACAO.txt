  INTEGRAÇÃO (PARA DEVELOPERS)
================================================================================

No seu skill ou comando:

  const AccountManager = require('./auto-panel/account-manager');
  const manager = new AccountManager();
  
  // Verificar se Google está conectado
  const google = manager.getAccount('google');
  if (!google?.connected) {
    return "Conecte Google em Integrações primeiro";
  }
  
  // Usar o token
  const token = manager.getToken('google');
  
  // Verificar expiração e renovar se necessário
  if (manager.isTokenExpired('google')) {
    // ... implementar refresh ...
  }
  
  // Usar token com API do Google
  const gmail = google.googleapis.gmail({ version: 'v1', auth: token });
  const emails = await gmail.users.messages.list({ userId: 'me' });
  
  return emails;
